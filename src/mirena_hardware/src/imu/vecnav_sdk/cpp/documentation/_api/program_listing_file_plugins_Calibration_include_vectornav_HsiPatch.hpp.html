    Patches212 = 212

class HsiPatch
{
public:
    enum class PatchType
    {
        NorthPatch,
        EastPatch
    };

    HsiPatch() = default;
    HsiPatch(const NumPatches numPatches, const uint16_t numDataPointsPerPatch) : _numPatches(numPatches), _numDataPointsPerPatch(numDataPointsPerPatch) {}

    int addData(const HsiDipole& dipole, const Ypr& ypr, PatchType patchType);

private:
    static float _dotAngle2cosErr(const float dotAngle) noexcept
    {
        float hFOV = Math::deg2rad(dotAngle / 2.0f);
        return std::cos(hFOV);
    }

    template <NumPatches N>
    static int _findPatch(const Vec3f& referenceVec)
    {
        const float cosErr = _dotAngle2cosErr(_patchDotAngles<N>);
        for (uint16_t i = 0; i < static_cast<uint16_t>(N); ++i)
        {
            const float dotProduct = Math::dot(referenceVec, _patchDefs<N>[i]);
            if (dotProduct >= cosErr) { return i; }
        }
        return -1;
    }

    static constexpr uint16_t _maxPatchCount = 212;
    std::array<uint16_t, _maxPatchCount> _northPatchCount{0};
    std::array<uint16_t, _maxPatchCount> _eastPatchCount{0};
    NumPatches _numPatches;
    uint16_t _numDataPointsPerPatch;

    template <NumPatches N>
    inline static const std::array<Vec3f, static_cast<uint16_t>(N)> _patchDefs{};
    template <NumPatches N>
    inline static const float _patchDotAngles{0};
};
